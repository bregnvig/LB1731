import { vi } from 'vitest';
...

it('should use spy', () => {
  const componentInstance = TestBed.createComponent(LoginFormComponent).componentInstance;
  componentInstance.email = 'email@email.com';
  componentInstance.password = 'password';
  const service = TestBed.inject(AuthService);
  const serviceSpy = vi.spyOn(service, 'login');

  componentInstance.login();

  expect(serviceSpy).toHaveBeenCalledWith('email@email.com', 'password');
  expect(serviceSpy).toHaveBeenCalledTimes(1);
});
Each spy should only live inside the specific test it's used for. This improves readability.
