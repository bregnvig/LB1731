retryWhen vs. retryWhenOnline

// Retries failed requests when the user is back online
export const retryWhenOnline = <T>() => pipe(
  retry<T>({
    delay: (error) => window.navigator.onLine
      ? throwError(() => error)
      : fromEvent(window, 'online')
  })
);

const playgrounds$ = this.service.playgrounds$.pipe(
  retryWhenOnline()
);
Conditional Retry Abstracts away complexity of handling retries based on network status
