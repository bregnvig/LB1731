Using takeUntilDestroyed() 

@Component({
  template: `
    <ul>
      @for (p of playgrounds; track p.id) {
        <li>{{ p.name }}</li>
      }
    </ul>
  `,
})
export class AppComponent {

  playgrounds?: Playground[];

  constructor() {
    inject(PlaygroundService).playgrounds$.pipe(
      takeUntilDestroyed()
    ).subscribe(playgrounds => this.playgrounds = playgrounds);
  }
