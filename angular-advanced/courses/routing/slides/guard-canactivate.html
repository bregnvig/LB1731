export class CanActivateRole implements CanActivate {
  // ...

  canActivate(
    route: ActivatedRouteSnapshot,
    state: RouterStateSnapshot): Observable<boolean | UrlTree> {
    return this.authService.roles$.pipe(
      map(roles => (roles || []).some(role => route.data.roles.includes(role))),
      map(isAuthorized => isAuthorized || this.router.createUrlTree(['/unauthorized']))
    );
  }
}
