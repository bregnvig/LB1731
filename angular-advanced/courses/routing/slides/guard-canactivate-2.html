export const hasRoleGuard = (
    route: ActivatedRouteSnapshot,
    state: RouterStateSnapshot): Observable<boolean | UrlTree> => {
    const router = inject(Router);
    return inject(AuthService).roles$.pipe(
      map(roles => (roles || []).some(role => route.data.roles.includes(role))),
      map(isAuthorized => isAuthorized || router.createUrlTree(['/unauthorized']))
    );
  };
