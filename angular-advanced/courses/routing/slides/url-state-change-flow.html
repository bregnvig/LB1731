ngOnInit(): void {
  this.filterControl.valueChanges.subscribe(
    term => this.router.navigate([], {queryParams: { term }})
  );
  
  combineLatest([
    this.playgrounds$,
    this.route.queryParams.pipe(map({term}) => new RegExp(term || '', 'i'))
  ]).subscribe(
    ([playgrounds, matcher]) => playgrounds.filter(({ name }) => matcher.test(name))
  )
}
