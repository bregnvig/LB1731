export const isAuthorizedGuard = (route: ActivatedRouteSnapshot) => {
  return inject(AuthService).roles$.pipe(
  map(roles => (roles || []).some(role => route.data.roles.includes(role))),
  map(isAuthorized => isAuthorized || this.router.createUrlTree(['/unauthorized']))
}
