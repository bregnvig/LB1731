@Injectable({  providedIn: 'root'})export class AarhusPlaygroundService {  playgrounds$ = of(aarhusPlaygrounds);   constructor() {    console.log('The AarhusPlaygroundService was just created');  }}
@Component({})export class WithDIComponent {   #locationService = inject(LocationService);  #service = inject(AarhusPlaygroundService);  playgrounds$: Observable<Playground[]> = this.#service.playgrounds$;  location$: Observable<Coordinate> = this.#locationService.location$; }
