Injector use case (prior to v14) 

export abstract class DialogComponent {

  constructor(
    private translateService: TranslateService, 
    private ref: ChangeDetectorRef
  ) {}

}

export class ConfirmDialogComponent extends DialogComponent {

  constructor(
    translateService: TranslateService, 
    ref: ChangeDetectorRef
  ) {
    super(translateService, ref)
  }

}

export abstract class DialogComponent {
  
  protected translate: TranslateService;
  protected ref: ChangeDetectorRef;
  protected elementRef: ElementRef;

  constructor(protected injector: Injector) {
    this.translate = this.injector.get(TranslateService);
    this.ref = this.injector.get(ChangeDetectorRef);
    this.elementRef = this.injector.get(ElementRef);
  }

}

export class ConfirmDialogComponent extends DialogComponent {

  constructor(injector: Injector) {
    super(injector)
  }
  
}
