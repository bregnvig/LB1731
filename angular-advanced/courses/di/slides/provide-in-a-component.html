@Component({
  selector: 'loop-root',
  template: `
    <leaflet [center]="center" [markers]="markers$" />
    <loop-sidebar [playgrounds]="playgrounds$ | async" (selected)="playground$.next($event)"/>
    @if (playground$ | async; as playground) {
      <loop-footer [playground]="playground"/>
    }
  `,
  providers: [
    {
      provide: PlaygroundService,
      useClass: AarhusService,
    }
  ]
  standalone: true,
  imports: [SidebarComponent, FooterComponent, AsyncPipe, LeafletModule]
})
