WHAT ABOUT THOSE THINGS THAT WEREN'T TRUE? (prior to v14)

@Optional() - If it is alright it's not there

export class RandomComponent {
  constructor(@Optional() public service: RandomService) { }
}

@SkipSelf() - Go find it somewhere. Just not here!

export class RandomComponent {
  constructor(@SkipSelf() public service: RandomService) { }
}

@Self() - Needs to be provided here

export class RandomComponent {
  constructor(@Self() public service: RandomService) { }
}

@Host() - Needs to be provided here or parent when using projection 

export class RandomComponent {
  constructor(@Host() public service: RandomService) { }
}
