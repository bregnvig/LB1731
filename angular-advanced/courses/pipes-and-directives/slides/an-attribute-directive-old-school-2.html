An Attribute directive - old school

@Directive({ selector: '[loopExpand]' })    @HostBinding('style.width') width?: string;   @HostListener('focus') onFocus() {    this.loopExpand && (this.width = `${this.ref.nativeElement.offsetWidth + 160}px`);  }   @HostListener('blur') onBlur() {    this.loopExpand && (this.width = `${this.ref.nativeElement.offsetWidth - 160}px`);  }}
