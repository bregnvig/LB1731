Effect Cleanup

@Component({ ...})
export class CounterComponent {
  count = signal(0);

  constructor() {
    const previousValue = signal<number | undefined>(undefined)
    effect(onCleanup => {
      const count = this.count();
      console.log(count, previousValue());
      onCleanup(() => previousValue.set(count));
    });
    setInterval(() => this.count.update(value => value + 1), 1000);
  }
}
