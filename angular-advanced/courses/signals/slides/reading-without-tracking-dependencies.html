Reading without tracking dependencies

Access a Signal's value without creating a dependency

import { ..., untracked } from '@angular/core';

@Component({...})
export class UserLocationComponent {
  user: Signal<User[]> = ...;
  location: Signal<Coordinate[]> = ...;

  constructor() {
    effect(() => {
      const location = this.location();
      const user = untracked(() => this.user());
      http.post(`api/location/${user.id}`, location)
    });
  }
  
}
