Signal Inputs - in Action

import { Component, input, computed, effect } from '@angular/core'; @Component({...})export class UserProfileComponent {    // Optional input with a default value  location = input<Coordinate | null>(null);   // Required input without an initial value  playground = input.required<Playground>();   // Derived value using computed  playgroundInfo = computed(() => {    const location = this.location();    if(location) {      const playground = this.playground();      const distance = this.ls.getDistance(location, playground),      return { ...playground, distance }     }    return undefined;  });   constructor(private ls: LocationService) {    // Monitoring changes with effect    effect(() => {      console.log('Location updated:', this.location());    });  }}
<p>Name: {{ playgroundInfo()?.name }}</p><p>Distance: {{ playgroundInfo()?.distance }}</p>
Notice the .required. Notice required does not have an initial value
