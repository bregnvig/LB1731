Signal Inputs - in Action

import { Component, computed, inject, input } from '@angular/core';

@Component({...})
export class UserProfileComponent {
  private ls = inject(LocationService);

  // Optional input with a default value
  location = input<Coordinate | null>(null);
  // Required input without an initial value
  playground = input.required<Playground>();

  // Derived value using computed
  playgroundInfo = computed(() => {
    const location = this.location();
    if (location) {
      const playground = this.playground();
      const distance = this.ls.getDistance(location, playground);
      return { ...playground, distance };
    }
    return undefined;
  });
}

<p>Name: {{ playgroundInfo()?.name }}</p>
<p>Distance: {{ playgroundInfo()?.distance }}</p>
Notice the .required. Notice required does not have an initial value. Demo: /signals/inputs (demo/src/app/signals/inputs) shows a decorator input next to a signal input.
