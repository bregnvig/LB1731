Function-based outputs - In action

import { Component, input, output, OutputEmitterRef } from '@angular/core';

@Component({
  selector: 'loop-playgrounds',
  template: `
	@for (playground of playgrounds(); track playground.id) {
	  <loop-playground 
		[playground]="playground"
		(click)="selectPlayground(playground)"/>  
	}
  `,
  ...
})
export class MyComponent {

  playgrounds = input.required<Playground[]>();
  
  select = output<Playground>();
  alias = output<Playground>({ alias: 'selectPlayground' });

  selectPlayground(playground: Playground): void {
    this.select.emit(playground);
    this.alias.emit(playground);
  }
}
<loop-playgrounds
  (select)="selectPlayground($event)"
  (selectPlayground)="selectPlayground($event)"
/>
