*ngComponentOutlet with input

Create a dynamic component using the component class

@Component({
  template: `
	<input type="text" [{ngModel}]="message"
	<ng-container *ngComponentOutlet="dynamicComponent; {inputs: {message} }" />
  `
}})
export class MyComponent {
  dynamicComponent = MyDynamicComponent;
  message = 'Hello';
}
