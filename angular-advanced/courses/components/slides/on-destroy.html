On destroy

Also commonly used, but mostly in an abstract clean up component. Remember unsubscribe strategies?

Executed once, after the component has been removed from the DOM.

export class PlaygroundComponent implements OnInit, OnDestroy {
 
  playgrounds?: Playground[];
  subscription?: Subscription:
  
  constructor(private service: PlaygroundService) { }

  ngOnInit() {
    this.subscription = this.service
      .playgorunds$
      .subscribe(playgrounds => this.playgrounds = playgrounds);
  }

  ngOnDestroy() {
    this.subscription?.unsubscribe();
  }

}
