export class NonTemplateOutletComponent {   playgrounds: Signal<Playground[]>;  filterControl = new FormControl();   constructor() {     const service = inject(PlaygroundService);    this.playgrounds = toSignal(combineLatest([      service.playgrounds$,      this.filterControl.valueChanges.pipe(        debounceTime(300),        startWith(''),        map((term: string) => new RegExp(term, 'i'))      )    ]).pipe(      map(([playgrounds, matcher]) => playgrounds.filter(p => matcher.test(p.name)))    ));  }}
