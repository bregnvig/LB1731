Register on change😵

  • Called once, when your custom control is being attached to the Angular form framework. Model driven or template driven
  • The parameter is a callback function for propagating changes in your control
export class AddressControlComponent {	  private propagateChange: ((_: any) => any) | undefined;    constructor() {    this.fg.valueChanges.pipe(      takeUntilDestroyed()    ).subscribe(value => this.propagateChange?(value));  }   registerOnChange(fn: any): void {    this.propagateChange = fn;  } }
