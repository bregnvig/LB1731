Register on touched🥴

  • Used to indicate your custom control has been touched
  • You'll need this if you use ng-touched &  ng-untouched for styling
export class AddressControlComponent implements OnInit {	  private propagateTouched: (() => void) | undefined;    onBlur() {    this.propagateTouched!();  }   registerOnTouched(fn: any): void {    this.propagateTouched = fn;  }}
