Set disabled state

  • Disables or enables your custom control
  • Being called whenever you use disable or enable in the parent control
export class AddressControlComponent implements OnInit {	  setDisabledState(isDisabled: boolean): void {    isDisabled ? this.fg.disable() : this.fg.enable();  }}
