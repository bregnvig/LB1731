@Service() since v22

  • Use @Service() and the service is provided in root. No providedIn needed

  • Use @Service({ autoProvided: false }) to provide it yourself

  • Inject with inject(). Constructor injection is not allowed

  • Added by the CLI: ng g s

@Service()
export class RandomLoggerService {

  #random = signal(Math.floor(Math.random() * 100));
  #logger = inject(LoggerService);

  constructor() {
    window.setInterval(() => {
      this.#random.set(Math.floor(Math.random() * 100));
      this.#logger.log('Generated new random number');
    }, 1000);
  }
}
