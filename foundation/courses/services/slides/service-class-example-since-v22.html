Service class example since v22

@Service()
export class RandomService {

  #random = signal(Math.floor(Math.random() * 100));

  constructor() {
    window.setInterval(() => this.#random.set(Math.floor(Math.random() * 100)), 1000);
  }

  get random(): number {
    return this.#random();
  }
}
Notice no ngOnInt or ngOnDestroy
