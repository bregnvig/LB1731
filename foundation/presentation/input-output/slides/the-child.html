@Component({
  selector: 'app-stopwatch',
  template: '{{seconds()}}',
})
export class StopwatchComponent {
  seconds = signal<number>(0);
  #intervalNo?: number;
  start() {
    this.#intervalNo = setInterval(() => this.seconds.update(value => value + 1), 1000);
  }
  stop() {
    clearInterval(this.#intervalNo);
  }
  reset() {
    this.stop();
    this.seconds.set(0);
  }
}
