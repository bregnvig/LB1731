A simple example

The service

@Injectable({providedIn: 'root'})
export class F1SimpleService {
  #http = inject(HttpClient);

  getDrivers(): Observable<any> {
    return this.#http.get(`https://api.openf1.org/v1/drivers?session_key=latest`);
  }
}
@Component({
  selector: 'app-simple-http-service',
  templateUrl: './simple-http-service.component.html',
})
export class SimpleHttpServiceComponent {

  protected drivers = signal<Driver[] | undefined>(undefined);
  #service = inject(F1SimpleService);

  constructor() {
    this.#service.getDrivers().subscribe(response => this.drivers.set(response.map((driver: any) => ({
      driverNumber: driver.driver_number.toString(),
      firstName: driver.first_name,
      photoURL: driver.headshot_url,
      lastName: driver.last_name,
    }))));
  }
}
