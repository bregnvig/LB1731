ngModelGroup directive

<fieldset ngModelGroup="name">
    <div>
        <label for="firstName">First name</label>
        <input id="firstName" ngModel name="firstName">
    </div>
    <div>
        <label for="lastName">Last name</label>
        <input id="lastName" ngModel name="lastName">
    </div>
</fieldset>

    name: {
        firstName: 'Flemming',
        lastName: 'Bregnvig'
    }
