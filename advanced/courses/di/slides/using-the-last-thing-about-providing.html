export class AppComponent implements OnInit {
  playgrounds$: Observable<Playground[]>;
  constrcutor(private service: PlaygroundService | PlaygroundService[]) { }
  
  ngOnInit() {
    const services = Array.isArray(this.service) ? this.service : [this.service]
    this.playgrounds$ = combineLatest(services.map(s => s.playgrounds$)).pipe(
      map((playgrounds: Playground[][]) => playgrounds.flat())
    );
  }
}
