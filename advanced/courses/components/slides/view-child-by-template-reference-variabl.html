export class ViewChildComponent implements OnInit {
  @ViewChild('myWatch', { static: true }) stopWatch?: StopWatchComponent;
  ngOnInit() {
    this.stopWatch?.start();
  }
}
export class ViewChildComponent {
  protected stopWatch = viewChild.required<StopWatchComponent>('myWatch');
  constructor() {
    afterNextRender(() => this.stopWatch()?.start())
  }
}
