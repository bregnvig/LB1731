Dynamic components v20😁

Use ViewContainerRef for creating component in the component class

It's doable in pre v20, but is a nightmare

@Component({
  selector: 'loop-my-component',
  template: `
    ...
    <ng-container #dynamicContainer />
    ...

  `
})
export class MyComponent {
  container = viewChild('dynamicContainer', { read: ViewContainerRef });
	
  constructor() {
    effect(() => {
      const container = this.container()
      container && container.createComponent(DisplayPlaygroundComponent, {
        bindings: [
          inputBinding('playground', () => playground),
          outputBinding('vote', (vote: 'up' | 'down') => {
            console.log(playground.name, vote);
          })
        ]
      })
    })
  }
}
