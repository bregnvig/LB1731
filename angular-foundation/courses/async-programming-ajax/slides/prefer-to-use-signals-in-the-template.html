Prefer to use signals in the template

Convert an observable to a signal

@Component({
  selector: 'app-map',
  template: `
	@for(playground of playgrounds(); track playground.id) {
		{{ playground.name }}
	} 
  `
})
export class MapComponent {
  
  protected playgrounds: Signal<Playground[] | undefined>;
  
  constructor() {
    const service = inject(PlaygroundService);
    this.playgrounds = toSignal(service.getPlaygrounds());
  }
}
This will automatically subscribe and unsubscribe from the observable
