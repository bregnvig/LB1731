Lazy loading

Loads the feature when needed - saves bandwidth and browser memory

const routes: Routes = [
  {
    path: 'orders',
    loadChildren: () => import('app/order/order.routes').then(m => m.OrderRoutes)
  },
  {
    path: 'invoice',
    component: InvoicesComponent
  },
  {
    path: '**',
    redirectTo: 'orders'
  }
]
