Adding or removing DOM elements

@Component({
  selector: 'app-ng-if',
  template: `
    <button (click)="toggle()">Toggle</button>
    @if(show) {
      <p>@if works!</p>
    }
  `
})
export class NgIfComponent {
  show = true;
  toggle() {
    this.show = !this.show;
  }
}

Since v17

@Component({
  selector: 'app-ng-if',
  template: `
    <button (click)="toggle()">Toggle</button>
    <p *ngIf="show">
      ngIf works!
    </p>
  `,
  imports: [NgIf]
})
export class NgIfComponent {
  show = true;
  toggle() {
    this.show = !this.show;
  }
}
