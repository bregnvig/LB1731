export class ProfileComponent {

  #model: Person = {
    firstName: 'Flemming', lastName: 'Bregnvig',
    favoriteColor: 'Blue', height: 182, shoeSize: 43
  };
  
  fg: FormGroup<PersonForm> = new FormGroup({
    firstName: new FormControl(this.#model.firstName, Validators.required),
    lastName: new FormControl(this.#model.lastName, Validators.required),
    favoriteColor: new FormControl(this.#model.favoriteColor),
    height: new FormControl(this.#model.height, [Validators.min(100), Validators.max(220)]),
  });
}
