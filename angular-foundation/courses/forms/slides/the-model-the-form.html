The model & the form

export interface Person {
  firstName: string;
  lastName: string;
  favoriteColor: string;
  height: number | null;
  shoeSize: number;
}

export class SignalFormComponent {

  #model = signal<Person>({
    firstName: 'Flemming', lastName: 'Bregnvig',
    favoriteColor: 'Blue', height: 182, shoeSize: 43 });

  protected form = form(this.#model, (person) => {
    required(person.firstName);
    required(person.lastName);
    min(person.height, 100);
    max(person.height, 220);
  });

}
