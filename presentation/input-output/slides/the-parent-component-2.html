@Component({
  selector: 'app-io-event',
  template: `
  <app-io-event-child
    (interval)="newsletterInterval = $event">
  </app-io-event-child>
  <p>
    Newsletter interval: {{newsletterInterval}}
  </p>
  `,
})
export class IOEventComponent {
  newsletterInterval?: string;
}
