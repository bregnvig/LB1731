@Injectable()

  • Use @Injectable() to provide a service by using providedIn 

  • Use @Injectable() to enabling injecting of a service into the annotated service

  • Always added by the CLI

@Injectable()
export class RandomLoggerService {

  #random = signal(Math.floor(Math.random() * 100));
  #logger = inject(LoggerService);

  constructor() {
    window.setInterval(() => {
      this.#random.set(Math.floor(Math.random() * 100));
      this.#logger.log('Generated new random number');
    }, 1000);
  }
}
